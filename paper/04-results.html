make_table <- function(models, col_names, blocks, extra_rows = list(), note = note_prev) {
  terms  <- unlist(lapply(blocks, `[[`, "terms"))
  labels <- unlist(lapply(blocks, `[[`, "labels"))
  idx    <- sapply(blocks, function(b) length(b$terms)); names(idx) <- sapply(blocks, `[[`, "name")

  coefs <- sapply(models, function(m) sapply(terms, function(t) cell(m, t)))
  rownames(coefs) <- labels

  fit_rows <- rbind(
    "N_i (observaciones)"            = sapply(models, nobs),
    "N_j (personas)"                 = sapply(models, function(m) ngrps(m)[["idencuesta"]]),
    "AIC"                            = sapply(models, function(m) fmt(AIC(m), 1)),
    "BIC"                            = sapply(models, function(m) fmt(BIC(m), 1)),
    "Var. intercepto (persona)"      = sapply(models, function(m) fmt(var_comp(m, "int"))),
    "Var. pendiente ola_num"         = sapply(models, function(m) fmt(var_comp(m, "time"))),
    "Var. pendiente aleatoria del WE interactuado" = sapply(models, function(m) {
                                          v <- var_comp(m, "sss"); ifelse(is.na(v), "", fmt(v, 4)) }),
    "Var. residual"                  = sapply(models, function(m) fmt(var_comp(m, "resid"))),
    "Ajuste singular"                = sapply(models, function(m) ifelse(isSingular(m), "sí", "no")),
    "Convergencia"                   = sapply(models, function(m)
                                          ifelse(is.null(m@optinfo$conv$lme4$messages), "ok", "advertencia")))

  if (length(extra_rows)) {
    fit_rows <- rbind(fit_rows, do.call(rbind, lapply(extra_rows, unname)))
  }

  out <- rbind(coefs, fit_rows)
  colnames(out) <- col_names
  idx <- c(idx, "Ajuste y efectos aleatorios" = nrow(fit_rows))

  kable(out, format = "html", align = c("l", rep("r", length(models))),
        format.args = list(big.mark = ".", decimal.mark = ",")) %>%
    kable_styling(bootstrap_options = c("striped", "condensed"), full_width = FALSE, font_size = 12) %>%
    pack_rows(index = idx) %>%
    footnote(general = note, general_title = "Nota: ", footnote_as_chunk = TRUE)
}

Modelling

Subjective social status (WE and BE) and income trajectory

One model per block of status (origin, own and children), without interactions: the income trajectory enters only as a main effect and the random-effects structure is (1 + ola_num | idencuesta) (random intercept and random time slope; no random slope for the WE status). The reference is Stable low.

  • Model 1: only the trend (linear and quadratic) and the trajectory.
  • Model 2 (base): adds the controls (ideology, sex, age and university education BE).
  • Models 3 to 5: to the base model, each block of status is added separately (origin, own and children, each with its WE and BE), not cumulatively.

# Helpers (same definitions as analysis/previ-sss.qmd). lmerTest is called with
# `::` so that it is not attached and the models above are not affected.
oc    <- lmerControl(optimizer = "bobyqa", optCtrl = list(maxfun = 2e5))
re_A  <- "(1 + ola_num | idencuesta)"
re_B  <- function(s) paste0("(1 + ola_num + ", s, "_we | idencuesta)")

fmt   <- function(x, digits = 3) formatC(x, format = "f", digits = digits, decimal.mark = ",")
star  <- function(p) ifelse(p < .001, "***", ifelse(p < .01, "**",
                     ifelse(p < .05, "*", ifelse(p < .1, "+", ""))))
p_fmt <- function(p) ifelse(p < .001, "< 0,001", fmt(p))

cell <- function(m, term) {
  s <- coef(summary(m))
  if (!term %in% rownames(s)) return("")
  paste0(fmt(s[term, 1]), " (", fmt(s[term, 2]), ")", star(s[term, 5]))
}

var_comp <- function(m, what) {
  v <- as.data.frame(VarCorr(m))
  switch(what,
    int   = v$vcov[v$grp == "idencuesta" & v$var1 == "(Intercept)" & is.na(v$var2)],
    time  = v$vcov[v$var1 == "ola_num" & is.na(v$var2) & !is.na(v$var1)],
    sss   = { x <- v$vcov[!is.na(v$var1) & endsWith(v$var1, "_we") & is.na(v$var2)]
              if (length(x)) x else NA_real_ },
    resid = sigma(m)^2)
}

# Table of models with texreg: one row per coefficient ("coef (se)***", single.row = TRUE),
# decimal comma, p-values from lmerTest (Satterthwaite).
#   blocks:     list of blocks (name, terms, labels) that group the rows
#   extra_rows: named list of extra footer rows (one value per model)
make_texreg <- function(models, col_names, blocks, extra_rows = list(), note = "", caption = NULL) {
  terms  <- unlist(lapply(blocks, `[[`, "terms"))
  labels <- unlist(lapply(blocks, `[[`, "labels"))

  # keep only the terms that appear in at least one model (texreg drops the rest)
  present <- sapply(terms, function(t) any(sapply(models, function(m) t %in% rownames(coef(summary(m))))))
  blk_n   <- sapply(blocks, function(b) sum(present[match(b$terms, terms)]))
  groups  <- setNames(split(seq_len(sum(blk_n)), rep(seq_along(blk_n), blk_n)),
                      sapply(blocks, `[[`, "name"))
  groups  <- groups[blk_n > 0]
  coef_map <- as.list(setNames(labels[present], terms[present]))

  gof_rows <- c(
    list("Ajuste singular" = sapply(models, function(m) ifelse(isSingular(m), "sí", "no")),
         "Convergencia"    = sapply(models, function(m)
           ifelse(is.null(m@optinfo$conv$lme4$messages), "ok", "advertencia"))),
    extra_rows)

  op <- options(OutDec = ","); on.exit(options(op))
  html <- texreg::htmlreg(
    models,
    custom.model.names = col_names,
    custom.coef.map = coef_map,
    groups = groups,
    custom.gof.rows = gof_rows,
    custom.note = note,
    single.row = TRUE,
    digits = 3,
    stars = c(0.001, 0.01, 0.05, 0.1), symbol = "+",
    include.loglik = FALSE,
    caption = caption, caption.above = TRUE,
    doctype = FALSE, html.tag = FALSE, head.tag = FALSE, body.tag = FALSE,
    center = FALSE, bold = 0)
  # thicker rule on the lines that separate the blocks of the table
  html <- gsub("border-top: 1px solid #000000", "border-top: 2px solid #000000", html, fixed = TRUE)
  for (g in names(groups)) {
    g_esc <- gsub("([][{}()+*^$|\\\\?.])", "\\\\\\1", g)
    html <- sub(paste0('<tr( style="[^"]*")?>(\\s*<td[^>]*>)', g_esc, "</td>"),
                paste0('<tr style="border-top: 2px solid #000000;">\\2', gsub("\\\\", "\\\\\\\\", g), "</td>"),
                html)
  }
  knitr::asis_output(html)
}

Income trajectory × subjective social status (WE and BE)

One model per status (origin, own and children). Each model includes the controls, the six main effects of status, the income trajectory and the interaction of the trajectory with the WE and the BE of that column’s status. The slope of the interacted WE status is random: (1 + ola_num + X_we | idencuesta). Each likelihood-ratio test compares the model with interactions against a base model with the same random slope but no interactions.

Same structure as the previous section, but the subjective social status is replaced by the subjective mobility children - origin (mov_exp_og), decomposed into its within-person (WE) and between-person (BE, uncentred) components. Subjective mobility x income trajectory

Same structure as the Income trajectory × subjective social status section, but the subjective social status is replaced by the subjective mobility children - origin (mov_exp_og, WE and BE). The model includes the controls, the mobility main effects (WE and BE), the income trajectory and the interaction of the trajectory with the mobility WE and BE. The slope of the mobility WE is random: (1 + ola_num + mov_exp_og_cwc | idencuesta). Each likelihood-ratio test compares the model with interactions against a base model with the same random slope but no interactions.

Code 
mv_we <- "mov_exp_og_cwc"
mv_be <- "mov_exp_og_mean"
main_m <- paste(mv_we, "+", mv_be, "+ traj_ingreso")
re_Bm  <- paste0("(1 + ola_num + ", mv_we, " | idencuesta)")

int_we_m <- paste0(mv_we, ":traj_ingreso")
int_be_m <- paste0(mv_be, ":traj_ingreso")

base_Am <- fit3(paste(ctrl3, "+", main_m))                 # without random slope of mobility
b_rsm   <- fit3(paste(ctrl3, "+", main_m), re_Bm)          # base with the same random slope
Tmi     <- fit3(paste(ctrl3, "+", main_m, "+", int_we_m, "+", int_be_m), re_Bm)
m_wem   <- fit3(paste(ctrl3, "+", main_m, "+", int_we_m), re_Bm)

lrt_mi <- c(
  conj = anova(b_rsm, Tmi)$`Pr(>Chisq)`[2],
  we   = anova(b_rsm, m_wem)$`Pr(>Chisq)`[2],
  rs   = anova(base_Am, b_rsm)$`Pr(>Chisq)`[2])
