# Equivalised income (square-root scale) --------------------------------------

# Select a reference IPC (e.g., December 2023 as base)
ipc_ref <- elsoc_long_2016_2023 %>%
  filter(ola == "2023") %>%
  summarise(ipc_ref = median(ipc, na.rm = TRUE)) %>%
  pull(ipc_ref)

  mutate(
    # Deflate household income to constant prices (base = 2023 Dec)
    inc_hh_real = m29_imp * (ipc_ref / ipc),
    
    # Equivalence scale: square-root of household size
    eq_size_sqrt = if_else(!is.na(n_hogar_r) & n_hogar_r > 0,
                           sqrt(as.numeric(n_hogar_r)),
                           NA_real_),
    
    # Equivalised household income
    inc_eq_real_sqrt = inc_hh_real / eq_size_sqrt,
    
    # Log equivalised income (for modelling)
    ln_inc_eq_real_sqrt = if_else(!is.na(inc_eq_real_sqrt) & inc_eq_real_sqrt > 0,
                                  log(inc_eq_real_sqrt),
                                  NA_real_)
  )

df_study1 <- df_study1 %>%
  mutate(mov_inter = ess_hijos - ess - ess_og) %>%
  group_by(idencuesta) %>%
  mutate(mov_inter_mean = mean(mov_inter),
         mov_inter_cwc  = mov_inter - mov_inter_mean) %>%
  ungroup()
# Children's SSS minus family-of-origin SSS: positive values = higher
# expected social mobility
df_study1 <- df_study1 %>%
  mutate(mov_exp_og = ess_hijos - ess_og) %>%
  group_by(idencuesta) %>%
  mutate(mov_exp_og_mean = mean(mov_exp_og),
         mov_exp_og_cwc  = mov_exp_og - mov_exp_og_mean) %>%
  ungroup()
# SSS trajectory within each person-wave: family of origin (t = 1), own (t = 2)
# and children (t = 3). The OLS slope of ess ~ t over these three points is the
# trajectory (positive = upward). NA if any of the three measures is missing
df_study1 <- df_study1 %>%
  rowwise() %>%
  mutate(ess_traj = {
    y <- c(ess_og, ess, ess_hijos)
    if (anyNA(y)) NA_real_ else unname(coef(lm(y ~ c(1, 2, 3)))[2])
  }) %>%
  ungroup()
